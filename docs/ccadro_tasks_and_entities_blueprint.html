@Transactional
public VehicleResponse createVehicle(VehicleCreateRequest request) {
    if (vehicleRepository.findByVehicleCode(request.getVehicleCode()).isPresent()) {
        throw new AppException(ErrorCode.VEHICLE_ALREADY_EXISTS);
    }
    if (vehicleRepository.findByPlateNumber(request.getPlateNumber()).isPresent()) {
        throw new AppException(ErrorCode.VEHICLE_ALREADY_EXISTS);
    }
    Vehicle vehicle = vehicleMapper.toEntity(request);
    vehicle.setStatus(VehicleStatus.AVAILABLE);
    return vehicleMapper.toResponse(vehicleRepository.save(vehicle));
}

@Transactional
public VehicleMaintenanceResponse scheduleMaintenance(VehicleMaintenanceRequest request) {
    Vehicle vehicle = findVehicleOrThrow(request.getVehicleId());
    VehicleMaintenance maintenance = vehicleMaintenanceMapper.toEntity(request);

    // BRULE-01: Bảo dưỡng khẩn cấp hoặc sửa chữa lập tức khóa xe sang MAINTENANCE
    if ("EMERGENCY".equalsIgnoreCase(request.getMaintenanceType())
            || "CORRECTIVE".equalsIgnoreCase(request.getMaintenanceType())) {
        vehicle.setStatus(VehicleStatus.MAINTENANCE);
        maintenance.setStatus("IN_PROGRESS");
    } else {
        maintenance.setStatus("SCHEDULED");
    }
    return vehicleMaintenanceMapper.toResponse(vehicleMaintenanceRepository.save(maintenance));
}

        Map<String, String> items,
        int failuresCount,
        boolean allPassed
) {
    public static ChecklistEvaluation from(PreTripInspectionRequest req) {
        Map<String, Boolean> raw = new LinkedHashMap<>();
        raw.put("tires", Boolean.TRUE.equals(req.getTirePressurePassed()));
        raw.put("brakes", Boolean.TRUE.equals(req.getBrakeSystemPassed()));
        raw.put("lights", Boolean.TRUE.equals(req.getLightingSystemPassed()));
        raw.put("fluids", Boolean.TRUE.equals(req.getFluidLevelPassed()));
        raw.put("cargo_securing", Boolean.TRUE.equals(req.getCargoSecuringPassed()));
        raw.put("emergency_kit", Boolean.TRUE.equals(req.getEmergencyKitPassed()));

        int failures = (int) raw.values().stream().filter(passed -> !passed).count();
        Map<String, String> formatted = new LinkedHashMap<>();
        raw.forEach((k, passed) -> formatted.put(k, passed ? "OK" : "FAILED"));

        return new ChecklistEvaluation(formatted, failures, failures == 0);
    }

    public InspectionStatus getInspectionStatus() {
        return allPassed ? InspectionStatus.PASSED : InspectionStatus.FAILED;
    }

    public VehicleStatus getTargetVehicleStatus() {
        return allPassed ? VehicleStatus.AVAILABLE : VehicleStatus.MAINTENANCE;
    }
}

@Component
public class GeometryMapper {

    private final GeometryFactory geometryFactory = new GeometryFactory(new PrecisionModel(), 4326);

    public Point toPoint(Double longitude, Double latitude) {
        if (longitude == null || latitude == null) return null;
        return geometryFactory.createPoint(new Coordinate(longitude, latitude));
    }

    @Named("toLongitude")
    public Double toLongitude(Point point) {
        return point != null ? point.getX() : null;
    }

    @Named("toLatitude")
    public Double toLatitude(Point point) {
        return point != null ? point.getY() : null;
    }
}

@Transactional
public DispatchOrderResponse createDispatchOrder(DispatchOrderCreateRequest request) {
    Shipment shipment = shipmentRepository.findById(request.getShipmentId())
            .orElseThrow(() -> new AppException(ErrorCode.SHIPMENT_NOT_FOUND));

    if ("DISPATCHED".equalsIgnoreCase(shipment.getStatus())
            || dispatchOrderRepository.existsByShipmentId(shipment.getId())) {
        throw new AppException(ErrorCode.SHIPMENT_ALREADY_DISPATCHED);
    }
    if (dispatchOrderRepository.existsByDispatchCode(request.getDispatchCode())) {
        throw new AppException(ErrorCode.DISPATCH_CODE_EXISTS);
    }

    // Gọi sang Asset Service (:8081) xác minh BRULE-01 (AVAILABLE) và BRULE-03 (Chống quá tải)
    validateVehicleAndDriverEligibility(request.getVehicleId(), request.getDriverId(), shipment.getCargoWeightTons());

    DispatchOrder dispatchOrder = shipmentMapper.toDispatchOrderEntity(request);
    shipment.setStatus("DISPATCHED");
    return shipmentMapper.toDispatchOrderResponse(dispatchOrderRepository.save(dispatchOrder));
}

private void validateVehicleAndDriverEligibility(UUID vehicleId, UUID driverId, BigDecimal cargoWeightTons) {
    Map<String, Object> vehicleData = fetchAssetMap("/api/v1/assets/vehicles/" + vehicleId, ErrorCode.VEHICLE_NOT_FOUND);
    String vehicleStatus = String.valueOf(vehicleData.get("status"));
    if (!"AVAILABLE".equalsIgnoreCase(vehicleStatus)) {
        throw new AppException(ErrorCode.VEHICLE_NOT_AVAILABLE); // BRULE-01
    }

    BigDecimal maxPayloadTons = new BigDecimal(String.valueOf(vehicleData.get("maxPayloadTons")));
    if (cargoWeightTons != null && cargoWeightTons.compareTo(maxPayloadTons) > 0) {
        throw new AppException(ErrorCode.INVALID_CARGO_WEIGHT);  // BRULE-03: Chặn quá tải cứng
    }
}

@Component
public class GeometryMapper {
    private final GeometryFactory geometryFactory = new GeometryFactory(new PrecisionModel(), 4326);

    @Named("toLineString")
    public LineString toLineString(List<double[]> waypoints) {
        if (waypoints == null || waypoints.size() < 2) return null;
        Coordinate[] coords = waypoints.stream()
                .map(wp -> new Coordinate(wp[1], wp[0])) // [lat, lon] -> Coordinate(lon, lat)
                .toArray(Coordinate[]::new);
        return geometryFactory.createLineString(coords);
    }
}

@Transactional
public List<RouteCandidateResponse> planRoutes(RoutePlanRequest req) {
    ShipmentDto shipment = fetchShipment(req.getShipmentId());
    VehicleDto vehicle = req.getVehicleId() != null ? fetchVehicle(req.getVehicleId()) : defaultVehicle();

    // BRULE-05: Chuẩn hóa tổng trọng số wCo2 + wTime + wDist = 1.0
    double rawCo2 = req.getCarbonWeight() != null ? req.getCarbonWeight().doubleValue() : properties.getRouting().getDefaultWeights().getCarbon();
    double rawTime = req.getTimeWeight() != null ? req.getTimeWeight().doubleValue() : properties.getRouting().getDefaultWeights().getTime();
    double rawDist = req.getDistanceWeight() != null ? req.getDistanceWeight().doubleValue() : properties.getRouting().getDefaultWeights().getCost();
    double weightSum = (rawCo2 + rawTime + rawDist) > 0 ? (rawCo2 + rawTime + rawDist) : 1.0;
    double wCo2 = rawCo2 / weightSum, wTime = rawTime / weightSum, wDist = rawDist / weightSum;

    List<RouteCandidate> savedCandidates = generateAndScoreCandidates(shipment, vehicle, wCo2, wTime, wDist);
    return savedCandidates.stream().map(routingMapper::toCandidateResponse).toList();
}

@Transactional
public RouteVersionResponse selectRoute(RouteSelectRequest req) {
    RouteCandidate candidate = routeCandidateRepository.findById(req.getCandidateId())
            .orElseThrow(() -> new ResourceNotFoundException("RouteCandidate not found: " + req.getCandidateId()));

    // 1. Ghi nhận biên bản quyết định của Điều phối viên (BRULE-09 Audit Trail)
    RouteDecision decision = routingMapper.toRouteDecision(req);
    routeDecisionRepository.save(decision);

    // 2. Vô hiệu hóa các RouteVersion cũ của cùng vận đơn (Immutability)
    List<RouteVersion> existingVersions = routeVersionRepository
            .findByShipmentIdOrderByVersionNumberDesc(req.getShipmentId());
    int nextVersionNum = existingVersions.isEmpty() ? 1 : existingVersions.get(0).getVersionNumber() + 1;
    existingVersions.forEach(v -> v.setActive(false));
    routeVersionRepository.saveAll(existingVersions);

    // 3. Khởi tạo RouteVersion mới ở trạng thái Active
    String reason = nextVersionNum == 1 ? "INITIAL_PLANNING" : "DYNAMIC_REROUTE";
    RouteVersion newVersion = routingMapper.toRouteVersion(candidate, nextVersionNum, reason);
    return routingMapper.toVersionResponse(routeVersionRepository.save(newVersion));
}

private void evaluateDeviationStateMachine(Trip trip, TelemetryEvent currentEvent, double currentDistMeters) {
    double yellowThreshold = properties.getTracking().getDeviation().getYellowThresholdMeters(); // 50m
    double redThreshold = properties.getTracking().getDeviation().getRedThresholdMeters();       // 200m
    int requiredPoints = properties.getTracking().getDeviation().getConsecutivePointsRequired(); // 3 điểm

    List<TelemetryEvent> recentValidEvents = telemetryEventRepository
            .findTop5ByTripIdAndQualityPassedTrueOrderByTimestampDesc(trip.getId());
    if (recentValidEvents.size() < requiredPoints) return;

    List<TelemetryEvent> window = recentValidEvents.subList(0, requiredPoints);
    boolean allRed = window.stream().allMatch(e -> e.getDistanceFromRouteMeters().doubleValue() >= redThreshold);
    boolean allYellow = window.stream().allMatch(e -> e.getDistanceFromRouteMeters().doubleValue() >= yellowThreshold);
    boolean allBackOnRoute = window.stream().allMatch(e -> e.getDistanceFromRouteMeters().doubleValue() <= yellowThreshold);

    List<DeviationAlert> openAlerts = deviationAlertRepository.findByTripIdAndStatus(trip.getId(), "OPEN");

    // Авто-Resolve: Tự động đóng cảnh báo khi 3 điểm liên tiếp quay về hành lang <= 50m
    if (allBackOnRoute && !openAlerts.isEmpty()) {
        openAlerts.forEach(alert -> alert.setStatus("RESOLVED"));
        deviationAlertRepository.saveAll(openAlerts);
        return;
    }

    if (allRed) {
        upsertDeviationAlert(trip, currentEvent, openAlerts, "RED", currentDistMeters, window);
        evaluateDynamicReroute(trip, currentEvent.getLocationGeom(), "DEVIATION_RED_ALERT");
    } else if (allYellow) {
        upsertDeviationAlert(trip, currentEvent, openAlerts, "YELLOW", currentDistMeters, window);
    }
}

@Transactional
public ExceptionResolution resolveIncident(UUID incidentId, ExceptionResolutionRequest request) {
    FieldIncident incident = fieldIncidentRepository.findById(incidentId)
            .orElseThrow(() -> new AppException(ErrorCode.INCIDENT_NOT_FOUND));

    String decision = request.getDecision().trim().toUpperCase();
    ExceptionResolution resolution = fieldIncidentMapper.toResolutionEntity(request, incidentId, decision);
    ExceptionResolution savedResolution = exceptionResolutionRepository.save(resolution);

    // BRULE-17: Phân nhánh xử lý minh bạch theo quyết định thẩm định của Giám sát viên
    switch (decision) {
        case "APPROVED_TRIGGER_REROUTE" -> incident.setStatus("APPROVED_REROUTE");
        case "APPROVED_EXCUSED"         -> incident.setStatus("APPROVED_EXCUSED");
        case "REJECTED"                 -> incident.setStatus("REJECTED");
        case "ESCALATE_TO_DEMERIT"      -> {
            incident.setStatus("ESCALATED_DEMERIT");
            int points = (request.getDemeritPoints() != null && request.getDemeritPoints() > 0)
                    ? request.getDemeritPoints() : 5;
            String category = request.getViolationCategory() != null
                    ? request.getViolationCategory() : "UNAUTHORIZED_DEVIATION";
            issueDemeritForIncident(incident, points, category, request.getReviewedBy(), request.getReviewNotes());
        }
        default -> throw new AppException(ErrorCode.INVALID_RESOLUTION_DECISION);
    }
    return savedResolution;
}

public EmergencyNotificationResponse ingestEmergencyEvent(EmergencyEventPayload payload) {
    // 1. Chống bão cảnh báo trùng lặp trong cửa sổ 300 giây trên cùng (eventType + referenceId)
    if (payload.getReferenceId() != null) {
        Instant since = Instant.now().minusSeconds(dedupWindowSeconds);
        Optional<EmergencyNotification> existing = repository
                .findFirstByEventTypeAndReferenceIdAndCreatedAtAfter(
                        payload.getEventType(), payload.getReferenceId(), since);
        if (existing.isPresent()) {
            return toResponse(existing.get());
        }
    }

    // 2. Lưu thông báo khẩn cấp vào DB ccadro_notification
    EmergencyNotification saved = repository.save(buildEntity(payload));
    EmergencyNotificationResponse response = toResponse(saved);

    // 3. Broadcast tức thời tới tất cả màn hình Điều phối viên đang mở qua SSE
    broadcastSseEvent("emergency-alert", response);
    return response;
}

    SagaStateLog saga = initSagaLog(req);
    boolean assetReserved = false;
    try {
        // Bước 1: Kiểm tra BRULE-01/03 & Khóa trạng thái Xe (IN_USE) + Tài xế (ON_DUTY) tại Asset Service (:8081)

public MultiFuelCalculationResult calculateMultiFuelCo2e(
        FuelPriceCatalog catalog, BigDecimal distanceKm, BigDecimal payloadTons,
        BigDecimal actualFuelCostVnd, BigDecimal invoiceUnitPriceVnd) {

    // 1. Tính lượng nhiên liệu ước tính theo định mức kỹ thuật & tải trọng (Tier 2 Modeled)
    BigDecimal ratePer100Km = catalog.getEmptyConsumptionPer100km()
            .add(catalog.getLoadedPenaltyPerTon100km().multiply(payloadTons));
    BigDecimal estimatedLiters = distanceKm.divide(BigDecimal.valueOf(100), 6, RoundingMode.HALF_UP)
            .multiply(ratePer100Km).setScale(2, RoundingMode.HALF_UP);

    // 2. Nếu có Hóa đơn tiền xăng (VNĐ) -> Quy đổi ra Số Lít thực tế (Tier 1 Primary)
    BigDecimal effectiveUnitPrice = (invoiceUnitPriceVnd != null && invoiceUnitPriceVnd.compareTo(BigDecimal.ZERO) > 0)
            ? invoiceUnitPriceVnd : catalog.getUnitPriceVnd();
    BigDecimal actualLiters = (actualFuelCostVnd != null && actualFuelCostVnd.compareTo(BigDecimal.ZERO) > 0)
            ? actualFuelCostVnd.divide(effectiveUnitPrice, 2, RoundingMode.HALF_UP)
            : estimatedLiters;

    // 3. Kiểm tra đối soát gian lận / thất thoát nhiên liệu (> 25% so với định mức GPS)
    BigDecimal variancePct = actualLiters.subtract(estimatedLiters).abs()
            .divide(estimatedLiters.max(BigDecimal.ONE), 4, RoundingMode.HALF_UP)
            .multiply(BigDecimal.valueOf(100)).setScale(2, RoundingMode.HALF_UP);
    boolean anomalyWarning = variancePct.compareTo(BigDecimal.valueOf(25.0)) > 0;

    // 4. Nhân hệ số phát thải đặc trưng của từng loại xăng/dầu theo chuẩn GLEC v3
    BigDecimal ttwCo2eKg = actualLiters.multiply(catalog.getTtwKgCo2PerUnit()).setScale(4, RoundingMode.HALF_UP);
    BigDecimal wttCo2eKg = actualLiters.multiply(catalog.getWttKgCo2PerUnit()).setScale(4, RoundingMode.HALF_UP);
    BigDecimal wtwCo2eKg = ttwCo2eKg.add(wttCo2eKg);

    return new MultiFuelCalculationResult(
            catalog.getFuelType(), effectiveUnitPrice, actualLiters, estimatedLiters,
            variancePct, anomalyWarning, ttwCo2eKg, wttCo2eKg, wtwCo2eKg);
